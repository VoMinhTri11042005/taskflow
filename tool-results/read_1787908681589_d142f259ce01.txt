    67→const columns: { id: Task['status']; label: string; color: string; bgColor: string }[] = [

    69→  { id: 'in_progress', label: 'Đang làm', color: 'text-amber-600', bgColor: 'bg-amber-50' },

    83→  google_sheet: { label: 'Google Sheet', icon: Table2, color: 'text-emerald-600' },
    84→  google_slide: { label: 'Google Slides', icon: Presentation, color: 'text-orange-600' },
    85→  google_form: { label: 'Google Form', icon: FileQuestion, color: 'text-violet-600' },

    87→};
    88→
    89→export function BoardView() {
    90→  const isMobile = useIsMobile();
    91→  const [mobileTab, setMobileTab] = useState<Task['status']>('todo');
    92→  const {
    93→    tasks, setTasks,
    94→    projects, setProjects,
    95→    members, setMembers,
    96→    selectedProjectId,
    97→  } = useAppStore();
    98→
    99→  const [dialogOpen, setDialogOpen] = useState(false);
   100→  const [editingTask, setEditingTask] = useState<Task | null>(null);
   101→  const [detailTask, setDetailTask] = useState<Task | null>(null);
   102→  const [detailOpen, setDetailOpen] = useState(false);
   103→
   104→  // Form state
   105→  const [formTitle, setFormTitle] = useState('');
   106→  const [formDesc, setFormDesc] = useState('');
   107→  const [formStatus, setFormStatus] = useState<Task['status']>('todo');
   108→  const [formPriority, setFormPriority] = useState<Task['priority']>('medium');
   109→  const [formProjectId, setFormProjectId] = useState('');
   110→  const [formAssigneeId, setFormAssigneeId] = useState('');
   111→  const [formDueDate, setFormDueDate] = useState('');
   112→
   113→  // Link form state
   114→  const [linkTitle, setLinkTitle] = useState('');
   115→  const [linkUrl, setLinkUrl] = useState('');
   116→  const [linkType, setLinkType] = useState<'google_doc' | 'google_sheet' | 'google_slide' | 'google_form' | 'other'>('google_doc');
   117→  const [linkDialogOpen, setLinkDialogOpen] = useState(false);
   118→
   119→  const fetchTasks = useCallback(async () => {
   120→    const params = new URLSearchParams();
   121→    if (selectedProjectId) params.set('projectId', selectedProjectId);
   122→    const res = await fetch(`/api/tasks?${params.toString()}`);
   123→    const data = await res.json();
   124→    setTasks(data);
   125→  }, [selectedProjectId, setTasks]);
   126→
   127→  useEffect(() => {
   128→    fetchTasks();
   129→    fetch('/api/projects').then((r) => r.json()).then(setProjects);
   130→    fetch('/api/members').then((r) => r.json()).then(setMembers);
   131→  }, [fetchTasks, setProjects, setMembers]);
   132→
   133→  function openCreateDialog(status?: Task['status']) {
   134→    setEditingTask(null);
   135→    setFormTitle('');
   136→    setFormDesc('');
   137→    setFormStatus(status || 'todo');
   138→    setFormPriority('medium');
   139→    setFormProjectId(selectedProjectId || (projects.find(p => p.status === 'active')?.id) || '');
   140→    setFormAssigneeId('');
   141→    setFormDueDate('');
   142→    setDialogOpen(true);
   143→  }
   144→
   145→  function openEditDialog(task: Task) {
   146→    setEditingTask(task);
   147→    setFormTitle(task.title);
   148→    setFormDesc(task.description || '');
   149→    setFormStatus(task.status);
   150→    setFormPriority(task.priority);
   151→    setFormProjectId(task.projectId);
   152→    setFormAssigneeId(task.assigneeId || '');
   153→    setFormDueDate(task.dueDate ? format(new Date(task.dueDate), 'yyyy-MM-dd') : '');
   154→    setDialogOpen(true);
   155→  }
   156→
   157→  async function handleSave() {
   158→    if (!formTitle.trim()) return;
   159→    try {
   160→      const body: Record<string, unknown> = {
   161→        title: formTitle,
   162→        description: formDesc || null,
   163→        status: formStatus,
   164→        priority: formPriority,
   165→        projectId: formProjectId,
   166→        assigneeId: formAssigneeId || null,
   167→        dueDate: formDueDate ? new Date(formDueDate).toISOString() : null,
   168→      };
   169→
   170→      if (editingTask) {
   171→        await fetch(`/api/tasks/${editingTask.id}`, {
   172→          method: 'PUT',
   173→          headers: { 'Content-Type': 'application/json' },
   174→          body: JSON.stringify(body),
   175→        });
   176→        toast.success('Đã cập nhật công việc');
   177→      } else {
   178→        await fetch('/api/tasks', {
   179→          method: 'POST',
   180→          headers: { 'Content-Type': 'application/json' },
   181→          body: JSON.stringify(body),
   182→        });
   183→        toast.success('Đã tạo công việc mới');
   184→      }
   185→      setDialogOpen(false);
   186→      fetchTasks();
   187→    } catch {
   188→      toast.error('Có lỗi xảy ra');
   189→    }
   190→  }
   191→
   192→  async function handleDeleteTask(id: string) {
   193→    try {
   194→      await fetch(`/api/tasks/${id}`, { method: 'DELETE' });
   195→      toast.success('Đã xóa công việc');
   196→      setDetailOpen(false);
   197→      fetchTasks();
   198→    } catch {
   199→      toast.error('Có lỗi xảy ra');
   200→    }
   201→  }
   202→
   203→  async function handleStatusChange(taskId: string, newStatus: Task['status']) {
   204→    try {
   205→      await fetch(`/api/tasks/${taskId}`, {
   206→        method: 'PUT',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({ status: newStatus }),
   209→      });
   210→      fetchTasks();
   211→    } catch {
   212→      toast.error('Không thể cập nhật trạng thái');
   213→    }
   214→  }
   215→
   216→  // Link handlers
   217→  async function handleAddLink() {
   218→    if (!detailTask || !linkUrl.trim()) return;
   219→    try {
   220→      await fetch(`/api/tasks/${detailTask.id}/links`, {
   221→        method: 'POST',
   222→        headers: { 'Content-Type': 'application/json' },
   223→        body: JSON.stringify({
   224→          title: linkTitle.trim() || linkTypeConfig[linkType].label,
   225→          url: linkUrl,
   226→          type: linkType,
   227→        }),
   228→      });
   229→      toast.success('Đã thêm liên kết');
   230→      setLinkTitle('');
   231→      setLinkUrl('');
   232→      setLinkDialogOpen(false);
   233→      fetchTasks();
   234→      // Refresh detail task
   235→      const res = await fetch(`/api/tasks/${detailTask.id}`);
   236→      const updated = await res.json();
   237→      setDetailTask(updated);
   238→    } catch {
   239→      toast.error('Có lỗi xảy ra');
   240→    }
   241→  }
   242→
   243→  async function handleDeleteLink(linkId: string) {
   244→    if (!detailTask) return;
   245→    try {
   246→      await fetch(`/api/links/${linkId}`, { method: 'DELETE' });
   247→      toast.success('Đã xóa liên kết');
   248→      fetchTasks();
   249→      const res = await fetch(`/api/tasks/${detailTask.id}`);
   250→      const updated = await res.json();
   251→      setDetailTask(updated);
   252→    } catch {
   253→      toast.error('Có lỗi xảy ra');
   254→    }
   255→  }
   256→
   257→  function openTaskDetail(task: Task) {
   258→    setDetailTask(task);
   259→    setDetailOpen(true);
   260→  }
   261→
   262→  function getDueDateInfo(dateStr: string) {
   263→    const date = new Date(dateStr);
   264→    if (isToday(date)) return { text: 'Hôm nay', className: 'text-amber-600' };
   265→    if (isPast(date)) return { text: 'Đã quá hạn', className: 'text-red-600' };
   266→    if (date <= addDays(new Date(), 3)) return { text: format(date, 'dd/MM', { locale: vi }), className: 'text-orange-600' };
   267→    return { text: format(date, 'dd/MM', { locale: vi }), className: 'text-muted-foreground' };
   268→  }
   269→
   270→  const filteredTasks = selectedProjectId
   271→    ? tasks.filter((t) => t.projectId === selectedProjectId)
   272→    : tasks;
   273→
   274→  const currentProject = projects.find((p) => p.id === selectedProjectId);
   275→
   276→  return (
   277→    <div className="flex flex-col h-full">
   278→      {/* Header */}
   279→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 pb-4 shrink-0">
   280→        <div className="min-w-0">
   281→          <h1 className="text-2xl font-bold tracking-tight truncate">
   282→            {currentProject ? currentProject.name : 'Bảng công việc'}
   283→          </h1>
   284→          <p className="text-muted-foreground">
   285→            {currentProject ? currentProject.description : 'Tất cả công việc của nhóm'}
   286→          </p>
   287→        </div>
   288→        <Button onClick={() => openCreateDialog('todo')}>
   289→          <Plus className="mr-2 h-4 w-4" />
   290→          Thêm việc
   291→        </Button>
   292→      </div>
   293→
   294→      {/* Mobile tab navigation */}
   295→      {isMobile && (
   296→        <div className="flex gap-1 overflow-x-auto pb-2 shrink-0 -mx-1 px-1">
   297→          {columns.map((col) => {
   298→            const count = filteredTasks.filter((t) => t.status === col.id).length;
   299→            return (
   300→              <button
   301→                key={col.id}
   302→                onClick={() => setMobileTab(col.id)}
   303→                className={cn(
   304→                  'flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-xs font-medium whitespace-nowrap transition-colors shrink-0',
   305→                  mobileTab === col.id
   306→                    ? cn(col.bgColor, col.color, 'shadow-sm')
   307→                    : 'text-muted-foreground hover:bg-muted'
   308→                )}
   309→              >
   310→                {col.label}
   311→                <span className={cn(
   312→                  'h-5 min-w-[20px] px-1.5 rounded-full flex items-center justify-center text-[10px] font-bold',
   313→                  mobileTab === col.id ? 'bg-white/80' : 'bg-muted'
   314→                )}>
   315→                  {count}
   316→                </span>
   317→              </button>
   318→            );
   319→          })}
   320→        </div>
   321→      )}
   322→
   323→      {/* Kanban Board - Desktop: all columns, Mobile: single column */}
   324→      <div className="flex-1 overflow-x-auto">
   325→        <div className={cn(
   326→          'flex gap-4 h-full pb-4',
   327→          !isMobile && 'min-w-max'
   328→        )}>
   329→          {(isMobile ? columns.filter(c => c.id === mobileTab) : columns).map((column) => {
   330→            const columnTasks = filteredTasks.filter((t) => t.status === column.id);
   331→            return (
   332→              <div
   333→                key={column.id}
   334→                className={cn(
   335→                  'flex-shrink-0 flex flex-col rounded-xl p-3',
   336→                  isMobile ? 'w-full flex-1' : 'w-80',
   337→                  column.bgColor
   338→                )}
   339→              >
   340→                {/* Column header */}
   341→                <div className="flex items-center justify-between mb-3">
   342→                  <div className="flex items-center gap-2">
   343→                    <h3 className={cn('text-sm font-semibold', column.color)}>
   344→                      {column.label}
   345→                    </h3>
   346→                    <Badge variant="secondary" className="text-xs h-5 px-1.5">
   347→                      {columnTasks.length}
   348→                    </Badge>
   349→                  </div>
   350→                  <Button
   351→                    variant="ghost"
   352→                    size="icon"
   353→                    className="h-7 w-7"
   354→                    onClick={() => openCreateDialog(column.id)}
   355→                  >
   356→                    <Plus className="h-4 w-4" />
   357→                  </Button>
   358→                </div>
   359→
   360→                {/* Task cards */}
   361→                <ScrollArea className="flex-1">
   362→                  <div className="space-y-2 pr-2">
   363→                    {columnTasks.map((task) => {
   364→                      const assignee = task.assignee;
   365→                      const project = task.project;
   366→                      const priority = priorityConfig[task.priority];
   367→                      const dueInfo = task.dueDate ? getDueDateInfo(task.dueDate) : null;
   368→                      const hasLinks = task.links && task.links.length > 0;
   369→
   370→                      return (
   371→                        <Card
   372→                          key={task.id}
   373→                          className="cursor-pointer hover:shadow-md transition-all group"
   374→                          onClick={() => openTaskDetail(task)}
   375→                        >
   376→                          <CardContent className="p-3 space-y-2">
   377→                            {/* Priority & project */}
   378→                            <div className="flex items-center justify-between">
   379→                              <Badge className={cn('text-[10px] px-1.5 py-0', priority?.className)}>
   380→                                {priority?.label}
   381→                              </Badge>
   382→                              {!selectedProjectId && project && (
   383→                                <div className="flex items-center gap-1">
   384→                                  <div
   385→                                    className="h-1.5 w-1.5 rounded-full"
   386→                                    style={{ backgroundColor: project.color }}
   387→                                  />\n                                  <span className="text-[10px] text-muted-foreground truncate max-w-[80px]">
   388→                                    {project.name}
   389→                                  </span>
   390→                                </div>
   391→                              )}
   392→                            </div>
   393→
   394→                            {/* Title */}
   395→                            <p className="text-sm font-medium leading-snug line-clamp-2">
   396→                              {task.title}
   397→                            </p>
   398→
   399→                            {/* Description preview */}
   400→                            {task.description && (
   401→                              <p className="text-xs text-muted-foreground line-clamp-1">
   402→                                {task.description}
   403→                              </p>
   404→                            )}
   405→
   406→                            {/* Link badges */}
   407→                            {hasLinks && (
   408→                              <div className="flex flex-wrap gap-1">
   409→                                {task.links!.slice(0, 3).map((link) => {
   410→                                  const config = linkTypeConfig[link.type] || linkTypeConfig.other;
   411→                                  const LIcon = config.icon;
   412→                                  return (
   413→                                    <a
   414→                                      key={link.id}
   415→                                      href={link.url}
   416→                                      target="_blank"
   417→                                      rel="noopener noreferrer"
   418→                                      onClick={(e) => e.stopPropagation()}
   419→                                      className="inline-flex"
   420→                                    >
   421→                                      <Badge
   422→                                        variant="outline"
   423→                                        className="text-[10px] h-5 px-1.5 gap-0.5 hover:bg-accent cursor-pointer transition-colors"
   424→                                      >
   425→                                        <LIcon className={cn('h-3 w-3', config.color)} />
   426→                                        {link.title}
   427→                                      </Badge>
   428→                                    </a>
   429→                                  );
   430→                                })}
   431→                                {task.links!.length > 3 && (
   432→                                  <Badge variant="outline" className="text-[10px] h-5 px-1.5">
   433→                                    +{task.links!.length - 3}
   434→                                  </Badge>
   435→                                )}
   436→                              </div>
   437→                            )}
   438→
   439→                            {/* Bottom row */}
   440→                            <div className="flex items-center justify-between pt-1">
   441→                              <div className="flex items-center gap-2">
   442→                                {dueInfo && (
   443→                                  <span className={cn('text-[10px] flex items-center gap-0.5', dueInfo.className)}>
   444→                                    <Calendar className="h-3 w-3" />
   445→                                    {dueInfo.text}
   446→                                  </span>
   447→                                )}
   448→                                {hasLinks && (
   449→                                  <LinkIcon className="h-3 w-3 text-muted-foreground" />
   450→                                )}
   451→                              </div>
   452→                              {assignee && (
   453→                                <div
   454→                                  className="h-6 w-6 rounded-full flex items-center justify-center text-[10px] font-bold text-white shrink-0"
   455→                                  style={{ backgroundColor: assignee.color }}
   456→                                  title={assignee.name}
   457→                                >
   458→                                  {assignee.name.charAt(0).toUpperCase()}
   459→                                </div>
   460→                              )}
   461→                            </div>
   462→                          </CardContent>
   463→                        </Card>
   464→                      );
   465→                    })}
   466→                    {columnTasks.length === 0 && (
   467→                      <div className="flex items-center justify-center py-8 text-sm text-muted-foreground/60">
   468→                        Kéo việc vào đây
   469→                      </div>
   470→                    )}
   471→                  </div>
   472→                </ScrollArea>
   473→              </div>
   474→            );
   475→          })}
   476→        </div>
   477→      </div>
   478→
   479→      {/* Create/Edit Task Dialog */}
   480→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   481→        <DialogContent className="max-w-lg">
   482→          <DialogHeader>
   483→            <DialogTitle>
   484→              {editingTask ? 'Chỉnh sửa công việc' : 'Tạo công việc mới'}
   485→            </DialogTitle>
   486→          </DialogHeader>
   487→          <div className="space-y-4 py-2">
   488→            <div className="space-y-2">
   489→              <Label>Tiêu đề *</Label>
   490→              <Input
   491→                value={formTitle}
   492→                onChange={(e) => setFormTitle(e.target.value)}
   493→                placeholder="Nhập tiêu đề công việc..."
   494→              />
   495→            </div>
   496→            <div className="space-y-2">
   497→              <Label>Mô tả</Label>
   498→              <Textarea
   499→                value={formDesc}
   500→                onChange={(e) => setFormDesc(e.target.value)}
   501→                placeholder="Mô tả chi tiết..."
   502→                rows={3}
   503→              />
   504→            </div>
   505→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   506→              <div className="space-y-2">
   507→                <Label>Trạng thái</Label>
   508→                <Select value={formStatus} onValueChange={(v) => setFormStatus(v as Task['status'])}>
   509→                  <SelectTrigger><SelectValue /></SelectTrigger>
   510→                  <SelectContent>
   511→                    {columns.map((c) => (
   512→                      <SelectItem key={c.id} value={c.id}>{c.label}</SelectItem>
   513→                    ))}
   514→                  </SelectContent>
   515→                </Select>
   516→              </div>
   517→              <div className="space-y-2">
   518→                <Label>Ưu tiên</Label>
   519→                <Select value={formPriority} onValueChange={(v) => setFormPriority(v as Task['priority'])}>
   520→                  <SelectTrigger><SelectValue /></SelectTrigger>
   521→                  <SelectContent>
   522→                    <SelectItem value="low">Thấp</SelectItem>
   523→                    <SelectItem value="medium">Trung bình</SelectItem>
   524→                    <SelectItem value="high">Cao</SelectItem>
   525→                    <SelectItem value="urgent">Khẩn cấp</SelectItem>
   526→                  </SelectContent>
   527→                </Select>
   528→              </div>
   529→            </div>
   530→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   531→              <div className="space-y-2">
   532→                <Label>Dự án</Label>
   533→                <Select value={formProjectId} onValueChange={setFormProjectId}>
   534→                  <SelectTrigger><SelectValue placeholder="Chọn dự án" /></SelectTrigger>
   535→                  <SelectContent>
   536→                    {projects.filter(p => p.status === 'active').map((p) => (
   537→                      <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   538→                    ))}
   539→                  </SelectContent>
   540→                </Select>
   541→              </div>
   542→              <div className="space-y-2">
   543→                <Label>Người thực hiện</Label>
   544→                <Select value={formAssigneeId} onValueChange={setFormAssigneeId}>
