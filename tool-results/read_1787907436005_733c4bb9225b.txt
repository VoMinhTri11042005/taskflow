    29→} from '@/components/ui/select';

    41→import {

    44→  DropdownMenuItem,
    45→  DropdownMenuTrigger,
    46→} from '@/components/ui/dropdown-menu';
    47→import {

    58→  X,
    59→} from 'lucide-react';

    64→
    65→const columns: { id: Task['status']; label: string; color: string; bgColor: string }[] = [
    66→  { id: 'todo', label: 'Cần làm', color: 'text-slate-600', bgColor: 'bg-slate-100' },
    67→  { id: 'in_progress', label: 'Đang làm', color: 'text-amber-600', bgColor: 'bg-amber-50' },
    68→  { id: 'review', label: 'Xem xét', color: 'text-violet-600', bgColor: 'bg-violet-50' },
    69→  { id: 'done', label: 'Hoàn thành', color: 'text-emerald-600', bgColor: 'bg-emerald-50' },
    70→];
    71→
    72→const priorityConfig: Record<string, { label: string; className: string }> = {
    73→  low: { label: 'Thấp', className: 'bg-slate-100 text-slate-700' },
    74→  medium: { label: 'TB', className: 'bg-amber-100 text-amber-700' },
    75→  high: { label: 'Cao', className: 'bg-orange-100 text-orange-700' },
    76→  urgent: { label: 'Gấp', className: 'bg-red-100 text-red-700' },
    77→};
    78→
    79→const linkTypeConfig: Record<string, { label: string; icon: React.ElementType; color: string }> = {
    80→  google_doc: { label: 'Google Doc', icon: FileText, color: 'text-blue-600' },
    81→  google_sheet: { label: 'Google Sheet', icon: Table2, color: 'text-emerald-600' },
    82→  other: { label: 'Liên kết', icon: LinkIcon, color: 'text-slate-600' },
    83→};
    84→
    85→export function BoardView() {
    86→  const isMobile = useIsMobile();
    87→  const [mobileTab, setMobileTab] = useState<Task['status']>('todo');
    88→  const {
    89→    tasks, setTasks,
    90→    projects, setProjects,
    91→    members, setMembers,
    92→    selectedProjectId,
    93→  } = useAppStore();
    94→
    95→  const [dialogOpen, setDialogOpen] = useState(false);
    96→  const [editingTask, setEditingTask] = useState<Task | null>(null);
    97→  const [detailTask, setDetailTask] = useState<Task | null>(null);
    98→  const [detailOpen, setDetailOpen] = useState(false);
    99→
   100→  // Form state
   101→  const [formTitle, setFormTitle] = useState('');
   102→  const [formDesc, setFormDesc] = useState('');
   103→  const [formStatus, setFormStatus] = useState<Task['status']>('todo');
   104→  const [formPriority, setFormPriority] = useState<Task['priority']>('medium');
   105→  const [formProjectId, setFormProjectId] = useState('');
   106→  const [formAssigneeId, setFormAssigneeId] = useState('');
   107→  const [formDueDate, setFormDueDate] = useState('');
   108→
   109→  // Link form state
   110→  const [linkTitle, setLinkTitle] = useState('');
   111→  const [linkUrl, setLinkUrl] = useState('');
   112→  const [linkType, setLinkType] = useState<'google_doc' | 'google_sheet' | 'other'>('google_doc');
   113→  const [linkDialogOpen, setLinkDialogOpen] = useState(false);
   114→
   115→  const fetchTasks = useCallback(async () => {
   116→    const params = new URLSearchParams();
   117→    if (selectedProjectId) params.set('projectId', selectedProjectId);
   118→    const res = await fetch(`/api/tasks?${params.toString()}`);
   119→    const data = await res.json();
   120→    setTasks(data);
   121→  }, [selectedProjectId, setTasks]);
   122→
   123→  useEffect(() => {
   124→    fetchTasks();
   125→    fetch('/api/projects').then((r) => r.json()).then(setProjects);
   126→    fetch('/api/members').then((r) => r.json()).then(setMembers);
   127→  }, [fetchTasks, setProjects, setMembers]);
   128→
   129→  function openCreateDialog(status?: Task['status']) {
   130→    setEditingTask(null);
   131→    setFormTitle('');
   132→    setFormDesc('');
   133→    setFormStatus(status || 'todo');
   134→    setFormPriority('medium');
   135→    setFormProjectId(selectedProjectId || (projects.find(p => p.status === 'active')?.id) || '');
   136→    setFormAssigneeId('');
   137→    setFormDueDate('');
   138→    setDialogOpen(true);
   139→  }
   140→
   141→  function openEditDialog(task: Task) {
   142→    setEditingTask(task);
   143→    setFormTitle(task.title);
   144→    setFormDesc(task.description || '');
   145→    setFormStatus(task.status);
   146→    setFormPriority(task.priority);
   147→    setFormProjectId(task.projectId);
   148→    setFormAssigneeId(task.assigneeId || '');
   149→    setFormDueDate(task.dueDate ? format(new Date(task.dueDate), 'yyyy-MM-dd') : '');
   150→    setDialogOpen(true);
   151→  }
   152→
   153→  async function handleSave() {
   154→    if (!formTitle.trim()) return;
   155→    try {
   156→      const body: Record<string, unknown> = {
   157→        title: formTitle,
   158→        description: formDesc || null,
   159→        status: formStatus,
   160→        priority: formPriority,
   161→        projectId: formProjectId,
   162→        assigneeId: formAssigneeId || null,
   163→        dueDate: formDueDate ? new Date(formDueDate).toISOString() : null,
   164→      };
   165→
   166→      if (editingTask) {
   167→        await fetch(`/api/tasks/${editingTask.id}`, {
   168→          method: 'PUT',
   169→          headers: { 'Content-Type': 'application/json' },
   170→          body: JSON.stringify(body),
   171→        });
   172→        toast.success('Đã cập nhật công việc');
   173→      } else {
   174→        await fetch('/api/tasks', {
   175→          method: 'POST',
   176→          headers: { 'Content-Type': 'application/json' },
   177→          body: JSON.stringify(body),
   178→        });
   179→        toast.success('Đã tạo công việc mới');
   180→      }
   181→      setDialogOpen(false);
   182→      fetchTasks();
   183→    } catch {
   184→      toast.error('Có lỗi xảy ra');
   185→    }
   186→  }
   187→
   188→  async function handleDeleteTask(id: string) {
   189→    try {
   190→      await fetch(`/api/tasks/${id}`, { method: 'DELETE' });
   191→      toast.success('Đã xóa công việc');
   192→      setDetailOpen(false);
   193→      fetchTasks();
   194→    } catch {
   195→      toast.error('Có lỗi xảy ra');
   196→    }
   197→  }
   198→
   199→  async function handleStatusChange(taskId: string, newStatus: Task['status']) {
   200→    try {
   201→      await fetch(`/api/tasks/${taskId}`, {
   202→        method: 'PUT',
   203→        headers: { 'Content-Type': 'application/json' },
   204→        body: JSON.stringify({ status: newStatus }),
   205→      });
   206→      fetchTasks();
   207→    } catch {
   208→      toast.error('Không thể cập nhật trạng thái');
   209→    }
   210→  }
   211→
   212→  // Link handlers
   213→  async function handleAddLink() {
   214→    if (!detailTask || !linkUrl.trim()) return;
   215→    try {
   216→      await fetch(`/api/tasks/${detailTask.id}/links`, {
   217→        method: 'POST',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({
   220→          title: linkTitle.trim() || linkTypeConfig[linkType].label,
   221→          url: linkUrl,
   222→          type: linkType,
   223→        }),
   224→      });
   225→      toast.success('Đã thêm liên kết');
   226→      setLinkTitle('');
   227→      setLinkUrl('');
   228→      setLinkDialogOpen(false);
   229→      fetchTasks();
   230→      // Refresh detail task
   231→      const res = await fetch(`/api/tasks/${detailTask.id}`);
   232→      const updated = await res.json();
   233→      setDetailTask(updated);
   234→    } catch {
   235→      toast.error('Có lỗi xảy ra');
   236→    }
   237→  }
   238→
   239→  async function handleDeleteLink(linkId: string) {
   240→    if (!detailTask) return;
   241→    try {
   242→      await fetch(`/api/links/${linkId}`, { method: 'DELETE' });
   243→      toast.success('Đã xóa liên kết');
   244→      fetchTasks();
   245→      const res = await fetch(`/api/tasks/${detailTask.id}`);
   246→      const updated = await res.json();
   247→      setDetailTask(updated);
   248→    } catch {
   249→      toast.error('Có lỗi xảy ra');
   250→    }
   251→  }
   252→
   253→  function openTaskDetail(task: Task) {
   254→    setDetailTask(task);
   255→    setDetailOpen(true);
   256→  }
   257→
   258→  function getDueDateInfo(dateStr: string) {
   259→    const date = new Date(dateStr);
   260→    if (isToday(date)) return { text: 'Hôm nay', className: 'text-amber-600' };
   261→    if (isPast(date)) return { text: 'Đã quá hạn', className: 'text-red-600' };
   262→    if (date <= addDays(new Date(), 3)) return { text: format(date, 'dd/MM', { locale: vi }), className: 'text-orange-600' };
   263→    return { text: format(date, 'dd/MM', { locale: vi }), className: 'text-muted-foreground' };
   264→  }
   265→
   266→  const filteredTasks = selectedProjectId
   267→    ? tasks.filter((t) => t.projectId === selectedProjectId)
   268→    : tasks;
   269→
   270→  const currentProject = projects.find((p) => p.id === selectedProjectId);
   271→
   272→  return (
   273→    <div className="flex flex-col h-full">
   274→      {/* Header */}
   275→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 pb-4 shrink-0">
   276→        <div className="min-w-0">
   277→          <h1 className="text-2xl font-bold tracking-tight truncate">
   278→            {currentProject ? currentProject.name : 'Bảng công việc'}
   279→          </h1>
   280→          <p className="text-muted-foreground">
   281→            {currentProject ? currentProject.description : 'Tất cả công việc của nhóm'}
   282→          </p>
   283→        </div>
   284→        <Button onClick={() => openCreateDialog('todo')}>
   285→          <Plus className="mr-2 h-4 w-4" />
   286→          Thêm việc
   287→        </Button>
   288→      </div>
   289→
   290→      {/* Mobile tab navigation */}
   291→      {isMobile && (
   292→        <div className="flex gap-1 overflow-x-auto pb-2 shrink-0 -mx-1 px-1">
   293→          {columns.map((col) => {
   294→            const count = filteredTasks.filter((t) => t.status === col.id).length;
   295→            return (
   296→              <button
   297→                key={col.id}
   298→                onClick={() => setMobileTab(col.id)}
   299→                className={cn(
   300→                  'flex items-center gap-1.5 rounded-lg px-3 py-1.5 text-xs font-medium whitespace-nowrap transition-colors shrink-0',
   301→                  mobileTab === col.id
   302→                    ? cn(col.bgColor, col.color, 'shadow-sm')
   303→                    : 'text-muted-foreground hover:bg-muted'
   304→                )}
   305→              >
   306→                {col.label}
   307→                <span className={cn(
   308→                  'h-5 min-w-[20px] px-1.5 rounded-full flex items-center justify-center text-[10px] font-bold',
   309→                  mobileTab === col.id ? 'bg-white/80' : 'bg-muted'
   310→                )}>
   311→                  {count}
   312→                </span>
   313→              </button>
   314→            );
   315→          })}
   316→        </div>
   317→      )}
   318→
   319→      {/* Kanban Board - Desktop: all columns, Mobile: single column */}
   320→      <div className="flex-1 overflow-x-auto">
   321→        <div className={cn(
   322→          'flex gap-4 h-full pb-4',
   323→          !isMobile && 'min-w-max'
   324→        )}>
   325→          {(isMobile ? columns.filter(c => c.id === mobileTab) : columns).map((column) => {
   326→            const columnTasks = filteredTasks.filter((t) => t.status === column.id);
   327→            return (
   328→              <div
   329→                key={column.id}
   330→                className={cn(
   331→                  'flex-shrink-0 flex flex-col rounded-xl p-3',
   332→                  isMobile ? 'w-full flex-1' : 'w-80',
   333→                  column.bgColor
   334→                )}
   335→              >
   336→                {/* Column header */}
   337→                <div className="flex items-center justify-between mb-3">
   338→                  <div className="flex items-center gap-2">
   339→                    <h3 className={cn('text-sm font-semibold', column.color)}>
   340→                      {column.label}
   341→                    </h3>
   342→                    <Badge variant="secondary" className="text-xs h-5 px-1.5">
   343→                      {columnTasks.length}
   344→                    </Badge>
   345→                  </div>
   346→                  <Button
   347→                    variant="ghost"
   348→                    size="icon"
   349→                    className="h-7 w-7"
   350→                    onClick={() => openCreateDialog(column.id)}
   351→                  >
   352→                    <Plus className="h-4 w-4" />
   353→                  </Button>
   354→                </div>
   355→
   356→                {/* Task cards */}
   357→                <ScrollArea className="flex-1">
   358→                  <div className="space-y-2 pr-2">
   359→                    {columnTasks.map((task) => {
   360→                      const assignee = task.assignee;
   361→                      const project = task.project;
   362→                      const priority = priorityConfig[task.priority];
   363→                      const dueInfo = task.dueDate ? getDueDateInfo(task.dueDate) : null;
   364→                      const hasLinks = task.links && task.links.length > 0;
   365→
   366→                      return (
   367→                        <Card
   368→                          key={task.id}
   369→                          className="cursor-pointer hover:shadow-md transition-all group"
   370→                          onClick={() => openTaskDetail(task)}
   371→                        >
   372→                          <CardContent className="p-3 space-y-2">
   373→                            {/* Priority & project */}
   374→                            <div className="flex items-center justify-between">
   375→                              <Badge className={cn('text-[10px] px-1.5 py-0', priority?.className)}>
   376→                                {priority?.label}
   377→                              </Badge>
   378→                              {!selectedProjectId && project && (
   379→                                <div className="flex items-center gap-1">
   380→                                  <div
   381→                                    className="h-1.5 w-1.5 rounded-full"
   382→                                    style={{ backgroundColor: project.color }}
   383→                                  />\n                                  <span className="text-[10px] text-muted-foreground truncate max-w-[80px]">
   384→                                    {project.name}
   385→                                  </span>
   386→                                </div>
   387→                              )}
   388→                            </div>
   389→
   390→                            {/* Title */}
   391→                            <p className="text-sm font-medium leading-snug line-clamp-2">
   392→                              {task.title}
   393→                            </p>
   394→
   395→                            {/* Description preview */}
   396→                            {task.description && (
   397→                              <p className="text-xs text-muted-foreground line-clamp-1">
   398→                                {task.description}
   399→                              </p>
   400→                            )}
   401→
   402→                            {/* Bottom row */}
   403→                            <div className="flex items-center justify-between pt-1">
   404→                              <div className="flex items-center gap-2">
   405→                                {dueInfo && (
   406→                                  <span className={cn('text-[10px] flex items-center gap-0.5', dueInfo.className)}>
   407→                                    <Calendar className="h-3 w-3" />
   408→                                    {dueInfo.text}
   409→                                  </span>
   410→                                )}
   411→                                {hasLinks && (
   412→                                  <LinkIcon className="h-3 w-3 text-muted-foreground" />
   413→                                )}
   414→                              </div>
   415→                              {assignee && (
   416→                                <div
   417→                                  className="h-6 w-6 rounded-full flex items-center justify-center text-[10px] font-bold text-white shrink-0"
   418→                                  style={{ backgroundColor: assignee.color }}
   419→                                  title={assignee.name}
   420→                                >
   421→                                  {assignee.name.charAt(0).toUpperCase()}
   422→                                </div>
   423→                              )}
   424→                            </div>
   425→                          </CardContent>
   426→                        </Card>
   427→                      );
   428→                    })}
   429→                    {columnTasks.length === 0 && (
   430→                      <div className="flex items-center justify-center py-8 text-sm text-muted-foreground/60">
   431→                        Kéo việc vào đây
   432→                      </div>
   433→                    )}
   434→                  </div>
   435→                </ScrollArea>
   436→              </div>
   437→            );
   438→          })}
   439→        </div>
   440→      </div>
   441→
   442→      {/* Create/Edit Task Dialog */}
   443→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   444→        <DialogContent className="max-w-lg">
   445→          <DialogHeader>
   446→            <DialogTitle>
   447→              {editingTask ? 'Chỉnh sửa công việc' : 'Tạo công việc mới'}
   448→            </DialogTitle>
   449→          </DialogHeader>
   450→          <div className="space-y-4 py-2">
   451→            <div className="space-y-2">
   452→              <Label>Tiêu đề *</Label>
   453→              <Input
   454→                value={formTitle}
   455→                onChange={(e) => setFormTitle(e.target.value)}
   456→                placeholder="Nhập tiêu đề công việc..."
   457→              />
   458→            </div>
   459→            <div className="space-y-2">
   460→              <Label>Mô tả</Label>
   461→              <Textarea
   462→                value={formDesc}
   463→                onChange={(e) => setFormDesc(e.target.value)}
   464→                placeholder="Mô tả chi tiết..."
   465→                rows={3}
   466→              />
   467→            </div>
   468→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   469→              <div className="space-y-2">
   470→                <Label>Trạng thái</Label>
   471→                <Select value={formStatus} onValueChange={(v) => setFormStatus(v as Task['status'])}>
   472→                  <SelectTrigger><SelectValue /></SelectTrigger>
   473→                  <SelectContent>
   474→                    {columns.map((c) => (
   475→                      <SelectItem key={c.id} value={c.id}>{c.label}</SelectItem>
   476→                    ))}
   477→                  </SelectContent>
   478→                </Select>
   479→              </div>
   480→              <div className="space-y-2">
   481→                <Label>Ưu tiên</Label>
   482→                <Select value={formPriority} onValueChange={(v) => setFormPriority(v as Task['priority'])}>
   483→                  <SelectTrigger><SelectValue /></SelectTrigger>
   484→                  <SelectContent>
   485→                    <SelectItem value="low">Thấp</SelectItem>
   486→                    <SelectItem value="medium">Trung bình</SelectItem>
   487→                    <SelectItem value="high">Cao</SelectItem>
   488→                    <SelectItem value="urgent">Khẩn cấp</SelectItem>
   489→                  </SelectContent>
   490→                </Select>
   491→              </div>
   492→            </div>
   493→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   494→              <div className="space-y-2">
   495→                <Label>Dự án</Label>
   496→                <Select value={formProjectId} onValueChange={setFormProjectId}>
   497→                  <SelectTrigger><SelectValue placeholder="Chọn dự án" /></SelectTrigger>
   498→                  <SelectContent>
   499→                    {projects.filter(p => p.status === 'active').map((p) => (
   500→                      <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   501→                    ))}
   502→                  </SelectContent>
   503→                </Select>
   504→              </div>
   505→              <div className="space-y-2">
   506→                <Label>Người thực hiện</Label>
   507→                <Select value={formAssigneeId} onValueChange={setFormAssigneeId}>
   508→                  <SelectTrigger><SelectValue placeholder="Chưa gán" /></SelectTrigger>
   509→                  <SelectContent>
   510→                    <SelectItem value="none">Chưa gán</SelectItem>
   511→                    {members.map((m) => (
   512→                      <SelectItem key={m.id} value={m.id}>{m.name}</SelectItem>
   513→                    ))}
   514→                  </SelectContent>
   515→                </Select>
   516→              </div>
   517→            </div>
   518→            <div className="space-y-2">
   519→              <Label>Hạn chót</Label>
   520→              <Input
   521→                type="date"
   522→                value={formDueDate}
   523→                onChange={(e) => setFormDueDate(e.target.value)}
   524→              />
   525→            </div>
   526→          </div>
   527→          <DialogFooter>
   528→            <DialogClose asChild>
   529→              <Button variant="outline">Hủy</Button>
   530→            </DialogClose>
   531→            <Button onClick={handleSave} disabled={!formTitle.trim()}>
   532→              {editingTask ? 'Cập nhật' : 'Tạo mới'}
   533→            </Button>
   534→          </DialogFooter>
   535→        </DialogContent>
   536→      </Dialog>
   537→
   538→      {/* Task Detail Dialog */}
   539→      <Dialog open={detailOpen} onOpenChange={setDetailOpen}>
   540→        <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto sm:max-h-[85vh]">
   541→          {detailTask && (
   542→            <>
   543→              <DialogHeader>
   544→                <div className="flex items-start justify-between">
   545→                  <DialogTitle className="text-lg pr-4">{detailTask.title}</DialogTitle>
   546→                  <DropdownMenu>
